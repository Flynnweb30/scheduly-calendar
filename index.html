Google Sheets Sync & CSV Template

Sync confirmed appointments directly to your Google Sheets document via an automated Webhook / Web App URL hook, or export pre-formatted CSV templates.

Whenever a visitor books an appointment, Scheduly triggers this URL hook with the pre-formatted row.

Open Google Sheets
Quick Google Sheets Hook Setup (60 seconds):
  1. Create a new spreadsheet at sheets.new
  2. Click Extensions > Apps Script in Google Sheets.
  3. Replace existing code with the snippet below and click Deploy > New deployment.
  4. Select type: Web app, set "Who has access" to Anyone, click Deploy and copy the Web app URL into the input above!
function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(["Booking ID", "Status", "Event Type", "Duration", "Date", "Time", "Timezone", "Host", "Attendee Name", "Attendee Email", "Location", "Notes", "Created At"]); } var data = JSON.parse(e.postData.contents); var row = data.csvRow || [data.bookingId, data.status, data.eventTitle, data.duration + "m", data.date, data.time, data.timezone, data.hostName, data.attendeeName, data.attendeeEmail, data.location, data.notes, data.createdAt]; sheet.appendRow(row); return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON); } }
